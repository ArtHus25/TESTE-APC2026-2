rock.setAnimation("rock");
rock.velocityY =  - 25;
rock.rotationSpeed = 2;

function draw() {
  background("skyblue");
  rock.velocityY = rock.velocityY + 1;
  if (rock.y > 370){
    rock.velocityY = - 25;
  }
  


  
  // update sprites
  
  drawSprites();
