INICIO  INP
        STA ELEMENTO
TESTE1  LDA ELEMENTO
        SUB MENOSUM
        BRZ FIM
TESTE2  LDA ELEMENTO
        SUB MAIOR
        ADD MENOSUM
        BRP ATUAL
NEXT    INP
        STA ELEMENTO
        BRA TESTE1
ATUAL   LDA ELEMENTO
        STA MAIOR
        BRA NEXT
FIM     LDA MAIOR
        OUT
        HLT
MAIOR   DAT -1
ELEMENTO DAT 
MENOSUM DAT -1
