INICIO  INP
        STA ELEMENTO
        LDA ELEMENTO
        STA MENOR
TESTE1  LDA ELEMENTO
        SUB MENOSUM
        BRZ FIM
TESTE2  LDA MENOR
        SUB ELEMENTO
        ADD MENOSUM
        BRP ATUAL
NEXT    INP
        STA ELEMENTO
        BRA TESTE1
ATUAL   LDA ELEMENTO
        STA MENOR
        BRA NEXT
FIM     LDA MENOR
        OUT
        HLT
MENOR   DAT 0
ELEMENTO DAT 0
MENOSUM DAT -1
